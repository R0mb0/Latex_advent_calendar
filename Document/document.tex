\newcommand{\drawStar}[3]{% #1=x, #2=y, #3=radius
	\begin{scope}[shift={(#1,#2)}]
		\pgfmathsetmacro{\rin}{0.42*#3}%
		\filldraw[fill=colAccent, draw=black, line width=0.65pt, line join=round]
		(90:#3) -- (126:\rin) -- (162:#3) -- (198:\rin) -- (234:#3)
		-- (270:\rin) -- (306:#3) -- (342:#3) -- (18:#3) -- (54:\rin) -- cycle;
	\end{scope}
}

\newcommand{\drawSnowflake}[3]{% #1=x, #2=y, #3=radius
	\begin{scope}[shift={(#1,#2)}]
		\pgfmathsetmacro{\rbranch}{0.58*#3}%
		\pgfmathsetmacro{\rsub}{0.30*#3}%
		\foreach \ang in {0,60,120,180,240,300} {
			\draw[black, line width=0.7pt, line cap=round] (0,0) -- (\ang:#3);
			\pgfmathsetmacro{\aone}{\ang+42}%
			\pgfmathsetmacro{\atwo}{\ang-42}%
			\draw[black, line width=0.55pt, line cap=round] (\ang:\rbranch) -- ++(\aone:\rsub);
			\draw[black, line width=0.55pt, line cap=round] (\ang:\rbranch) -- ++(\atwo:\rsub);
		}
		\filldraw[fill=colBg, draw=black, line width=0.5pt] (0,0) circle (0.045);
	\end{scope}
}

\newcommand{\drawBow}[2]{% #1=center y, #2=scale
	\begin{scope}[shift={(0,#1)}, scale=#2]
		% Ribbon tails
		\filldraw[fill=colAccent, draw=black, line width=1.15pt, line join=round]
		(-0.14,-0.04) -- (-0.92,-0.92) -- (-0.50,-0.80) -- (-0.36,-1.05) -- (0,-0.14) -- cycle;
		\filldraw[fill=colAccent, draw=black, line width=1.15pt, line join=round]
		(0.14,-0.04) -- (0.92,-0.92) -- (0.50,-0.80) -- (0.36,-1.05) -- (0,-0.14) -- cycle;
		% Left loop
		\filldraw[fill=colAccent, draw=black, line width=1.25pt, line join=round]
		(0,0) .. controls (-0.58,0.76) and (-1.55,0.68) .. (-1.45,0.06)
		.. controls (-1.35,-0.46) and (-0.55,-0.34) .. (0,0) -- cycle;
		\draw[black, line width=0.8pt, line cap=round]
		(-0.24,0.06) .. controls (-0.64,0.18) and (-0.94,0.12) .. (-1.08,0.02);
		% Right loop
		\filldraw[fill=colAccent, draw=black, line width=1.25pt, line join=round]
		(0,0) .. controls (0.58,0.76) and (1.55,0.68) .. (1.45,0.06)
		.. controls (1.35,-0.46) and (0.55,-0.34) .. (0,0) -- cycle;
		\draw[black, line width=0.8pt, line cap=round]
		(0.24,0.06) .. controls (0.64,0.18) and (0.94,0.12) .. (1.08,0.02);
		% Center knot
		\filldraw[fill=colAccent, draw=black, line width=1.25pt, rounded corners=2.5pt]
		(-0.24,-0.22) rectangle (0.24,0.22);
	\end{scope}
}
